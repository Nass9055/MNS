logic: 'Future Location → Trap → Trigger → Execution' }; } // ========================================================== // CREATE EVENT PLANS // ========================================================== createEventPlans() { const event = this.map.event; if (!event || !event.active || !Number.isFinite(event.strikePrice)) { return; } const direction = event.direction || 'BUY'; const entry = event.strikePrice; const technicalSL = direction === 'BUY' ? entry - (this.map.range.buffer || entry * 0.001) : entry + (this.map.range.buffer || entry * 0.001); const spread = this.map.getSpread(); if (!Number.isFinite(spread)) return; const executionSL = direction === 'BUY' ? technicalSL - spread : technicalSL + spread; const risk = Math.abs(entry - executionSL); if (!risk) return; const tp1 = direction === 'BUY' ? entry + risk * 1.5 : entry - risk * 1.5; const tp2 = direction === 'BUY' ? entry + risk * CONFIG.MIN_RR : entry - risk * CONFIG.MIN_RR; this.plans.push({ id: `MNS-${++this.counter}`, name: direction === 'BUY' ? Confidence (0-100) const confidence = MNSUtils.clamp(plan.confidence || 50); // 3. Risk / Reward (0-100) const rr = plan.rr || CONFIG.MIN_RR; const rrScore = MNSUtils.clamp((rr / CONFIG.MIN_RR) * 70, 0, 100); // 4. Timeliness (0-100) const distPct = MNSUtils.pctDistance(this.map.midPrice, plan.entry); const timeliness = MNSUtils.clamp(100 - (distPct * 20), 10, 100); // 5. Location Quality (0-100) const locationQuality = MNSUtils.clamp(plan.locationQuality || 50); // Weighted Total const totalScore = Math.round( marketAlignment * weights.MARKET_ALIGNMENT + confidence * weights.CONFIDENCE + rrScore * weights.RISK_REWARD + timeliness * weights.TIMELINESS + locationQuality * weights.LOCATION_QUALITY ); return { ...plan, score: totalScore, metrics: { marketAlignment, confidence, rrScore, timeliness, locationQuality } }; } // ========================================================== // PROCESS & RANK // ========================================================== process() { const validPlans = this.plans.filter(p => p.state === CONFIG.STATUS.ARMED); this.scoredPlans = validPlans .map(p => this.scorePlan(p)) .filter(p => p.score >= CONFIG.MIN_PLAN_SCORE) .sort((a, b) => b.score - a.score); return this.scoredPlans.slice(0, CONFIG.TOP_N_PLANS); } // ========================================================== // GENERATE REPORT // ========================================================== generateReport() { const topPlans = this.process(); const quality = this.map.calculateQuality(); const routeInfo = this.map.getDirectionAndRoute(); return { version: CONFIG.VERSION, timestamp: new Date().toISOString(), market: { midPrice: this.map.midPrice, direction: routeInfo.primary, confidence: routeInfo.confidence, qualityScore: quality, route: routeInfo.route }, topPlans }; } }
